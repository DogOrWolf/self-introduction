# 体育考试管理系统
- 安装部署

  ```
   # 1. 更新服务器软件包
  yum update -y

  # 2. 安装 Node.js 和 npm（CentOS 推荐用 nvm 管理版本，避免版本冲突）
  # 安装 nvm
  curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.1/install.sh | bash
  # 刷新环境变量
  source ~/.bashrc
  # 安装 Node.js 14.x
  nvm install 14.21.3
  nvm use 14.21.3
  
  # 3. 验证安装
  node -v  # 输出 v14.21.3 即为成功
  npm -v   # 输出对应版本即可
  
  # 4. 安装 Nginx
  yum install nginx -y
  # 启动 Nginx 并设置开机自启
  systemctl start nginx
  systemctl enable nginx
  ```

- 基础环境配置

  ```
  # 1. 编辑 Nginx 配置文件
  vim /etc/nginx/conf.d/sport-exam-platform.conf
  
  # 2. 粘贴以下配置（核心：指向 dist 文件夹，解决 Vue 路由刷新404问题）
  server {
  listen 80;
  # 若有域名，填写域名；无则填服务器 IP
  server_name 你的服务器IP/域名;
  
      # 指向打包后的 dist 目录
      root /data/project/sport-exam-platform/dist;
      index index.html;
  
      # 解决 Vue 路由（history 模式）刷新404问题
      location / {
          try_files $uri $uri/ /index.html;
      }
  
      # 静态资源缓存配置（可选，提升访问速度）
      location ~* \.(js|css|png|jpg|jpeg|gif|ico)$ {
          expires 7d;
          add_header Cache-Control "public, max-age=604800";
      }
  
      # 禁止访问 .env 等配置文件（安全）
      location ~ /\.env {
          deny all;
      }
  }
  
  # 3. 保存配置并验证 Nginx 配置是否正确
  nginx -t
  
  # 4. 重启 Nginx 使配置生效
  systemctl restart nginx
  ```
